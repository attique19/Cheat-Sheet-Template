JavaScript Cheat Sheet

  • Variables

    Declare values using var, let, or const.

    let name = "Attique";
const age = 21;
var city = "Mianwali";

  • Data Types

    JS has 7 primitive types.

    let str = "Hello";           // String
let num = 42;                // Number
let bool = true;             // Boolean
let nothing = null;          // Null
let undef = undefined;       // Undefined
let sym = Symbol("id");      // Symbol
let big = 9007199254740991n; // BigInt

  • Template Literals

    Embed variables inside strings using backticks.

    let name = "Attique";
let msg = `Hello, ${name}! Welcome.`;
console.log(msg); // Hello, Attique! Welcome.

  • Conditionals

    Run code based on a condition.

    let score = 75;
if (score >= 90) {
  console.log("A Grade");
} else if (score >= 60) {
  console.log("B Grade");
} else {
  console.log("Fail");
}

  • Loops

    Repeat code multiple times.

    // for loop
for (let i = 0; i < 3; i++) {
  console.log(i); // 0, 1, 2
}

// while loop
let x = 0;
while (x < 3) {
  console.log(x);
  x++;
}

  • Functions

    Reusable blocks of code.

    // Regular function
function greet(name) {
  return `Hello, ${name}!`;
}

// Arrow function
const add = (a, b) => a + b;

console.log(greet("Attique")); // Hello, Attique!
console.log(add(3, 4));        // 7

  • Arrays

    Store multiple values in one variable.

    let fruits = ["apple", "mango", "banana"];

console.log(fruits[0]);     // apple
fruits.push("grape");       // add to end
fruits.pop();               // remove from end
console.log(fruits.length); // 3

  • Array Methods — map, filter, reduce

    Transform arrays without changing the original.

    let nums = [1, 2, 3, 4, 5];

let doubled = nums.map(n => n * 2);        // [2, 4, 6, 8, 10]
let evens = nums.filter(n => n % 2 === 0); // [2, 4]
let sum = nums.reduce((acc, n) => acc + n, 0); // 15

  • Objects

    Store related data as key-value pairs.

    let student = {
  name: "Attique",
  age: 21,
  university: "UMW"
};

console.log(student.name);   // Attique
console.log(student["age"]); // 21
student.city = "Mianwali";   // add new key

  • Spread Operator

    Copy or merge arrays/objects without mutation.

    let arr1 = [1, 2, 3];
let arr2 = [...arr1, 4, 5];   // [1, 2, 3, 4, 5]

let obj1 = { a: 1 };
let obj2 = { ...obj1, b: 2 }; // { a: 1, b: 2 }

  • Destructuring

    Unpack values from arrays or objects easily.

    // Array destructuring
let [first, second] = ["apple", "mango"];

// Object destructuring
let { name, age } = { name: "Attique", age: 21 };

console.log(first); // apple
console.log(name);  // Attique

  • Default Parameters

    Set fallback values for function parameters.

    function greet(name = "Guest") {
  return `Hello, ${name}!`;
}

console.log(greet());          // Hello, Guest!
console.log(greet("Attique")); // Hello, Attique!

  • String Methods

    Useful built-in methods for strings.

    let str = "  Hello World  ";

str.trim();                 // "Hello World"
str.toLowerCase();          // "  hello world  "
str.toUpperCase();          // "  HELLO WORLD  "
str.includes("World");      // true
str.split(" ");             // ["", "", "Hello", "World", "", ""]
str.replace("World", "JS"); // "  Hello JS  "

  • Ternary Operator

    A short one-line if/else.

    let age = 18;
let result = age >= 18 ? "Adult" : "Minor";
console.log(result); // Adult

  • Nullish Coalescing (??)

    Return the right side only if left is null or undefined.

    let username = null;
let display = username ?? "Guest";
console.log(display); // Guest

let score = 0;
let final = score ?? 10;
console.log(final); // 0 (not 10, because 0 is NOT null/undefined)

  • Optional Chaining (?.)

    Safely access nested properties without errors.

    let user = { profile: { name: "Attique" } };

console.log(user.profile?.name);  // Attique
console.log(user.address?.city);  // undefined (no error!)

  • Promises and Async/Await

    Handle asynchronous operations cleanly.

    // Promise
fetch("https://api.example.com/data")
  .then(res => res.json())
  .then(data => console.log(data))
  .catch(err => console.log(err));

// Async/Await (cleaner version)
async function getData() {
  try {
    let res = await fetch("https://api.example.com/data");
    let data = await res.json();
    console.log(data);
  } catch (err) {
    console.log(err);
  }
}

  • for...of and for...in

    Loop through arrays and objects.

    let fruits = ["apple", "mango"];

// for...of → loops through VALUES (use for arrays)
for (let fruit of fruits) {
  console.log(fruit); // apple, mango
}

let student = { name: "Attique", age: 21 };

// for...in → loops through KEYS (use for objects)
for (let key in student) {
  console.log(key, student[key]); // name Attique, age 21
}

  • Error Handling

    Catch and handle errors gracefully.

    try {
  let result = riskyFunction(); // might throw error
  console.log(result);
} catch (error) {
  console.log("Error caught:", error.message);
} finally {
  console.log("This always runs");
}

  • localStorage

    Save data in the browser that persists after refresh.

    // Save data
localStorage.setItem("username", "Attique");

// Get data
let user = localStorage.getItem("username");
console.log(user); // Attique

// Remove data
localStorage.removeItem("username");

// Clear everything
localStorage.clear();
